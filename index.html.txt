        const questionText = document.getElementById('question-text');
        const optionsContainer = document.getElementById('options-container');
        const nextBtn = document.getElementById('next-btn');
        const resetBtn = document.getElementById('reset-btn');
        const playAgainBtn = document.getElementById('play-again-btn');
        const resultContainer = document.getElementById('result-container');
        const quizContainer = document.querySelector('.quiz-container');
        const progressContainer = document.querySelector('.progress-container');
        const controls = document.querySelector('.controls');
        const currentQuestionElement = document.getElementById('current-question');
        const questionCounter = document.getElementById('question-counter');
        const progressFill = document.getElementById('progress-fill');
        const finalScoreElement = document.getElementById('final-score');
        const resultMessage = document.getElementById('result-message');
        const resultFeedback = document.getElementById('result-feedback');
        const highScoreElement = document.getElementById('high-score');
        const totalCorrectElement = document.getElementById('total-correct');
        const gamesPlayedElement = document.getElementById('games-played');
        const scoreCircle = document.getElementById('score-circle');

        // Lấy dữ liệu từ localStorage
        let highScore = localStorage.getItem('highScore') || 0;
        let totalCorrect = localStorage.getItem('totalCorrect') || 0;
        let gamesPlayed = localStorage.getItem('gamesPlayed') || 0;

        // Cập nhật hiển thị thống kê
        function updateStats() {
            highScoreElement.textContent = highScore;
            totalCorrectElement.textContent = totalCorrect;
            gamesPlayedElement.textContent = gamesPlayed;
        }

        // Lưu dữ liệu vào localStorage
        function saveStats() {
            localStorage.setItem('highScore', highScore);
            localStorage.setItem('totalCorrect', totalCorrect);
            localStorage.setItem('gamesPlayed', gamesPlayed);
        }

        // Khởi tạo ứng dụng
        function initApp() {
            updateStats();
            resetBtn.addEventListener('click', startNewGame);
            nextBtn.addEventListener('click', nextQuestion);
            playAgainBtn.addEventListener('click', startNewGame);
            startNewGame();
        }

        // Bắt đầu trò chơi mới
        function startNewGame() {
            // Ẩn kết quả, hiển thị câu hỏi
            resultContainer.style.display = 'none';
            quizContainer.style.display = 'block';
            progressContainer.style.display = 'block';
            controls.style.display = 'flex';
            
            // Reset biến
            currentQuestionIndex = 0;
            score = 0;
            userAnswers = [];
            questionsAnswered = 0;
            gameInProgress = true;
            
            // Chọn ngẫu nhiên 10 câu hỏi
            selectedQuestions = getRandomQuestions(10);
            
            // Hiển thị câu hỏi đầu tiên
            displayQuestion(currentQuestionIndex);
            
            // Cập nhật thanh tiến trình
            updateProgress();
            
            // Tăng số lượt chơi
            gamesPlayed++;
            saveStats();
            updateStats();
        }

        // Lấy ngẫu nhiên các câu hỏi
        function getRandomQuestions(num) {
            // Tạo bản sao của ngân hàng câu hỏi
            const shuffled = [...questionBank].sort(() => Math.random() - 0.5);
            
            // Lấy số câu hỏi cần thiết
            const selected = shuffled.slice(0, num);
            
            // Xáo trộn đáp án trong mỗi câu hỏi
            selected.forEach(question => {
                const originalOptions = [...question.options];
                const originalCorrect = question.correct;
                const correctAnswer = originalOptions[originalCorrect];
                
                // Tạo mảng mới cho các đáp án đã xáo trộn
                const shuffledOptions = [...originalOptions].sort(() => Math.random() - 0.5);
                
                // Tìm vị trí mới của đáp án đúng
                const newCorrectIndex = shuffledOptions.indexOf(correctAnswer);
                
                // Cập nhật câu hỏi
                question.options = shuffledOptions;
                question.correct = newCorrectIndex;
            });
            
            return selected;
        }

        // Hiển thị câu hỏi
        function displayQuestion(index) {
            if (index >= selectedQuestions.length) {
                showResults();
                return;
            }
            
            const question = selectedQuestions[index];
            
            // Cập nhật số câu hỏi
            questionCounter.textContent = index + 1;
            currentQuestionElement.textContent = index + 1;
            
            // Hiển thị nội dung câu hỏi
            questionText.textContent = question.question;
            
            // Xóa các lựa chọn cũ
            optionsContainer.innerHTML = '';
            
            // Tạo các lựa chọn mới
            const letters = ['A', 'B', 'C', 'D'];
            question.options.forEach((option, optionIndex) => {
                const optionElement = document.createElement('div');
                optionElement.className = 'option';
                optionElement.innerHTML = `
                    <div class="option-letter">${letters[optionIndex]}</div>
                    <div class="option-text">${option}</div>
                `;
                
                // Thêm sự kiện click
                optionElement.addEventListener('click', () => selectAnswer(optionIndex, question.correct));
                
                optionsContainer.appendChild(optionElement);
            });
            
            // Ẩn nút "Câu tiếp theo" cho đến khi chọn đáp án
            nextBtn.style.display = 'none';
        }

        // Chọn đáp án
        function selectAnswer(selectedIndex, correctIndex) {
            // Nếu câu hỏi đã được trả lời, không làm gì
            if (userAnswers[currentQuestionIndex] !== undefined) return;
            
            const options = document.querySelectorAll('.option');
            const letters = ['A', 'B', 'C', 'D'];
            
            // Đánh dấu câu hỏi đã được trả lời
            userAnswers[currentQuestionIndex] = selectedIndex;
            questionsAnswered++;
            
            // Kiểm tra đáp án
            const isCorrect = selectedIndex === correctIndex;
            
            // Hiển thị kết quả
            if (isCorrect) {
                options[selectedIndex].classList.add('correct');
                score++;
            } else {
                options[selectedIndex].classList.add('incorrect');
                options[correctIndex].classList.add('correct-answer');
            }
            
            // Vô hiệu hóa tất cả các lựa chọn
            options.forEach(option => {
                option.style.pointerEvents = 'none';
            });
            
            // Hiển thị nút "Câu tiếp theo"
            nextBtn.style.display = 'flex';
            
            // Cập nhật tổng số câu đúng
            if (isCorrect) {
                totalCorrect++;
                saveStats();
                updateStats();
            }
        }

        // Chuyển đến câu hỏi tiếp theo
        function nextQuestion() {
            currentQuestionIndex++;
            
            if (currentQuestionIndex < selectedQuestions.length) {
                displayQuestion(currentQuestionIndex);
                updateProgress();
            } else {
                showResults();
            }
        }

        // Cập nhật thanh tiến trình
        function updateProgress() {
            const progress = ((currentQuestionIndex + 1) / selectedQuestions.length) * 100;
            progressFill.style.width = `${progress}%`;
        }

        // Hiển thị kết quả
        function showResults() {
            gameInProgress = false;
            
            // Ẩn phần câu hỏi, hiển thị kết quả
            quizContainer.style.display = 'none';
            progressContainer.style.display = 'none';
            controls.style.display = 'none';
            resultContainer.style.display = 'block';
            
            // Cập nhật điểm số
            finalScoreElement.textContent = score;
            
            // Cập nhật điểm cao nhất
            if (score > highScore) {
                highScore = score;
                saveStats();
                updateStats();
            }
            
            // Cập nhật đồ thị vòng tròn điểm số
            const percentage = (score / selectedQuestions.length) * 100;
            scoreCircle.style.background = `conic-gradient(var(--primary-green) 0% ${percentage}%, #e0e0e0 ${percentage}% 100%)`;
            
            // Hiển thị thông điệp phù hợp
            let message = "";
            let feedback = "";
            
            if (score === selectedQuestions.length) {
                message = "Xuất sắc! Bạn là Chiến Binh Xanh thực thụ!";
                feedback = "Bạn có hiểu biết tuyệt vời về bảo vệ môi trường. Hãy tiếp tục phát huy và chia sẻ kiến thức với mọi người nhé!";
            } else if (score >= selectedQuestions.length * 0.7) {
                message = "Rất tốt! Bạn là Chiến Binh Xanh giỏi!";
                feedback = "Bạn có kiến thức tốt về bảo vệ môi trường. Hãy tiếp tục học hỏi và thực hành nhiều hơn nữa!";
            } else if (score >= selectedQuestions.length * 0.5) {
                message = "Khá tốt! Bạn đang trên đường trở thành Chiến Binh Xanh!";
                feedback = "Bạn đã có một số hiểu biết về bảo vệ môi trường. Hãy tiếp tục tìm hiểu thêm và áp dụng vào cuộc sống hàng ngày!";
            } else {
                message = "Cố gắng hơn nhé! Bảo vệ môi trường là việc của mọi người!";
                feedback = "Đừng nản lòng! Hãy tìm hiểu thêm về cách phân loại rác, tiết kiệm điện nước và bảo vệ cây xanh. Mỗi hành động nhỏ đều có ý nghĩa!";
            }
            
            resultMessage.textContent = message;
            resultFeedback.textContent = feedback;
        }

        // Khởi chạy ứng dụng
        window.onload = initApp;
